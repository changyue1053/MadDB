cmake_minimum_required(VERSION 3.20)
project(MadDBOptionalLocalRuntime LANGUAGES C CXX)
option(MADDB_WITH_MADOBS_RUNTIME "Explicit local synthetic runtime verification" OFF)
if(NOT MADDB_WITH_MADOBS_RUNTIME)
  message(FATAL_ERROR "This separate composition requires explicit ON")
endif()
find_package(MadObs 0.1.0 EXACT CONFIG REQUIRED PATHS "${MADOBS_PREFIX}/lib/cmake/MadObs" NO_DEFAULT_PATH)
if(NOT TARGET MadObs::runtime)
  message(FATAL_ERROR "Installed static runtime required")
endif()
add_library(maddb_installed SHARED IMPORTED)
set_target_properties(maddb_installed PROPERTIES IMPORTED_LOCATION "${MADDB_PREFIX}/lib/libmaddb.so"
  INTERFACE_INCLUDE_DIRECTORIES "${MADDB_PREFIX}/include")
add_executable(consumer main.cpp)
target_compile_features(consumer PRIVATE cxx_std_17)
target_compile_options(consumer PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(consumer PRIVATE maddb_installed MadObs::runtime)
