cmake_minimum_required(VERSION 3.20)
project(MadInstalledCohostApi LANGUAGES C CXX)
option(COHOST_OBS_API "Opt in to the installed API-only MadObs observer" OFF)
if(NOT COHOST_OBS_API)
  message(FATAL_ERROR "API-only cohost requires explicit ON")
endif()
find_package(MadDB REQUIRED CONFIG PATHS "${DB_PREFIX}" NO_DEFAULT_PATH)
if(NOT IS_ABSOLUTE "${LINK_PREFIX}" OR NOT EXISTS "${LINK_PREFIX}/lib/libmadlink.a")
  message(FATAL_ERROR "Missing relocated MadLink")
endif()
add_library(PilotMadLink STATIC IMPORTED)
set_target_properties(PilotMadLink PROPERTIES
  IMPORTED_LOCATION "${LINK_PREFIX}/lib/libmadlink.a"
  INTERFACE_INCLUDE_DIRECTORIES "${LINK_PREFIX}/include")
add_executable(cohost main.cc)
set_target_properties(cohost PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
target_compile_options(cohost PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(cohost PRIVATE MadDB::MadDB PilotMadLink)
set_target_properties(cohost PROPERTIES BUILD_RPATH "${DB_PREFIX}/lib")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/core-paths"
  CONTENT "$<TARGET_FILE:MadDB::MadDB>\n$<TARGET_FILE:PilotMadLink>\n")
find_package(MadObs 0.1.0 EXACT REQUIRED CONFIG PATHS "${OBS_PREFIX}" NO_DEFAULT_PATH)
if(NOT TARGET MadObs::api OR TARGET MadObs::runtime)
  message(FATAL_ERROR "Expected only the MadObs API target")
endif()
target_include_directories(cohost PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/adapters")
target_link_libraries(cohost PRIVATE MadObs::api)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/obs-api-path"
  CONTENT "$<TARGET_PROPERTY:MadObs::api,INTERFACE_INCLUDE_DIRECTORIES>")
